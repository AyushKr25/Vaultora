Question 1: Show that the users table has a PRIMARY KEY on id and NOT NULL on email.

SQL Statement:
DESC users;

Output:
+----------------+--------------+------+-----+-------------------+-------------------+
| Field          | Type         | Null | Key | Default           | Extra             |
+----------------+--------------+------+-----+-------------------+-------------------+
| id             | int          | NO   | PRI | NULL              | auto_increment    |
| email          | varchar(255) | NO   | UNI | NULL              |                   |
| password_hash  | varchar(255) | NO   |     | NULL              |                   |
| first_name     | varchar(100) | NO   |     | NULL              |                   |
| last_name      | varchar(100) | NO   |     | NULL              |                   |
| role_id        | int          | NO   | MUL | 2                 |                   |
| is_active      | tinyint(1)   | YES  |     | 1                 |                   |
+----------------+--------------+------+-----+-------------------+-------------------+

Explanation: The id column is the PRIMARY KEY (PRI) and email is UNIQUE (UNI).
             first_name, last_name, email are NOT NULL.

| ACC2024ADMIN1  |                  6 |
| ACC202602724519|                  4 |

3.7 Complex Queries based on Triggers
================================================================================

Question 1: Create a BEFORE INSERT trigger to prevent negative balance.

SQL Statement:
DELIMITER //
CREATE TRIGGER trg_before_account_insert
BEFORE INSERT ON bank_accounts
FOR EACH ROW
BEGIN
    IF NEW.balance < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Balance cannot be negative';
    END IF;
END//
DELIMITER ;

Output:
Trigger created successfully.

Testing the trigger:
INSERT INTO bank_accounts (account_number, user_id, balance)
VALUES ('TEST001', 1, -500);

Output:
ERROR 1644 (45000): Balance cannot be negative

Explanation: The trigger prevents inserting an account with a negative balance.

Question 2: Create an AFTER INSERT trigger to log deposits into audit_logs.

SQL Statement:
DELIMITER //
CREATE TRIGGER trg_after_deposit
AFTER INSERT ON transactions
FOR EACH ROW
BEGIN
    IF NEW.transaction_type = 'deposit' THEN
        INSERT INTO audit_logs (user_id, action, entity_type, entity_id, new_values, created_at)
        VALUES (
            (SELECT user_id FROM bank_accounts WHERE id = NEW.account_id),
            'TRANSACTION_DEPOSIT',
            'transaction',
            NEW.id,
            JSON_OBJECT('amount', NEW.amount, 'ref', NEW.transaction_ref),
            NOW()
        );
    END IF;
END//
DELIMITER ;

Output:
Trigger created successfully.

Explanation: Every time a deposit transaction is inserted, it is automatically
             logged into the audit_logs table.

3.8 Complex Queries based on Cursors

Question 1: Cursor to display all user names and their email.

SQL Statement:
DELIMITER //
CREATE PROCEDURE show_all_users()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_name VARCHAR(200);
    DECLARE v_email VARCHAR(255);

    DECLARE user_cursor CURSOR FOR
        SELECT CONCAT(first_name, ' ', last_name), email FROM users;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    OPEN user_cursor;

    read_loop: LOOP
        FETCH user_cursor INTO v_name, v_email;
        IF done THEN
            LEAVE read_loop;
        END IF;
        SELECT v_name AS Name, v_email AS Email;
    END LOOP;

    CLOSE user_cursor;
END//
DELIMITER ;

CALL show_all_users();

CREATE PROCEDURE show_high_balance_accounts()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_acc VARCHAR(20);

    DECLARE acc_cursor CURSOR FOR
        SELECT account_number, balance FROM bank_accounts WHERE balance > 10000;

        IF done THEN
            LEAVE read_loop;

    END LOOP;

    CLOSE acc_cursor;
END//

Question 3: Cursor to calculate and display total balance per user.

SQL Statement:
DELIMITER //
CREATE PROCEDURE show_user_total_balance()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_name VARCHAR(200);
    DECLARE v_total DECIMAL(15,2);

    DECLARE balance_cursor CURSOR FOR
        SELECT CONCAT(u.first_name, ' ', u.last_name) AS name,
               SUM(b.balance) AS total
        FROM users u
        JOIN bank_accounts b ON u.id = b.user_id
        GROUP BY u.id;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    OPEN balance_cursor;

    read_loop: LOOP
        FETCH balance_cursor INTO v_name, v_total;
        IF done THEN
            LEAVE read_loop;
        END IF;
        SELECT v_name AS User_Name, v_total AS Total_Balance;
    END LOOP;

    CLOSE balance_cursor;
END//
DELIMITER ;

CALL show_user_total_balance();

Output:

+---------------+---------------+
| Admin User    |      11050.00 |
| John Doe      |      20000.00 |
| Jane Smith    |      75000.00 |
